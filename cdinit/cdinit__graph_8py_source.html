17def parse_service_file(file_path):
18 """
19 Parse a dinit service file and extract dependencies and service type.
20
21 Args:
22 file_path (str): Path to the service file
23
24 Returns:
25 dict: A dictionary with service name, type and its dependencies
26 """
27 dependencies = {
28 'depends-on': [],
29 'depends-ms': [],
30 'waits-for': [],
31 'after': [],
32 'before': [],
33 'depends-on.d': [],
34 'depends-ms.d': [],
35 'waits-for.d': []
36 }
37
38 service_type = 'process' # Default service type
39 has_parameters = False # Flag to detect if the service supports arguments
40
41 try:
42 with open(file_path, 'r', encoding='utf-8') as f:
43 content = f.read()
44 except (UnicodeDecodeError, PermissionError):
45 # Skip binary or unreadable files
46 sys.stderr.write(f'Warning: Could not read file {file_path}\n')
47 service_name = os.path.basename(file_path)
48 service_name = service_name.split('@')[0] if '@' in service_name else service_name
49 return {'name': service_name, 'type': service_type, 'dependencies': dependencies, 'has_parameters': has_parameters}
50
51 current_line = 0
52 lines = content.splitlines()
53
54 while current_line < len(lines):
55 line = lines[current_line].strip()
56 current_line += 1
57
58 # Skip empty lines and comments
59 if not line or line.startswith('#'):
60 continue
61
62 # Handle meta-commands (lines starting with @)
63 if line.startswith('@'):
64 # For @include and @include-opt, we would need to process included files
65 # For now, we'll just skip them
66 continue
67
68 # Check if this line contains parameter references (like $1, $2, $3 etc.)
69 if re.search(r'\$\d+', line):
70 has_parameters = True
71
72 # Match property patterns
73 # Format: property = value or property: value
74 match = re.match(r'^([a-zA-Z0-9._-]+)\s*([:=])\s*(.*)$', line)
75 if match:
76 prop_name = match.group(1).strip()
77 value = match.group(3).strip()
78
79 # Remove trailing comments
80 if '#' in value:
81 value = value[:value.index('#')].strip()
82
83 # Handle service type
84 if prop_name == 'type':
85 service_type = value.split()[0] if value.split() else 'process'
86
87 # Handle dependency properties
88 if prop_name in dependencies:
89 # Handle multi-line values if they exist
90 full_value = value
91 while current_line < len(lines) and lines[current_line].startswith(' '):
92 continued_line = lines[current_line].strip()
93 if continued_line:
94 full_value += ' ' + continued_line
95 current_line += 1
96
97 # Add dependencies (space-separated values)
98 deps = full_value.split()
99 # Filter out parameters after @ symbol in dependency names
100 # Also filter out any parameter placeholders like $1, $2, etc.
101 filtered_deps = []
102 for dep in deps:
103 # Remove @ parameters
104 dep = dep.split('@')[0] if '@' in dep else dep
105 # Skip parameter placeholders like $1, $2, etc.
106 if not dep.startswith('$'):
107 filtered_deps.append(dep)
108 dependencies[prop_name].extend(filtered_deps)
109
110 # Extract service name without parameters (before @ symbol)
111 service_name = os.path.basename(file_path)
112 service_name = service_name.split('@')[0] if '@' in service_name else service_name
113
114 return {
115 'name': service_name,
116 'type': service_type,
117 'dependencies': dependencies,
118 'has_parameters': has_parameters
119 }

122def get_services_from_directory(directory):
123 """
124 Get all service files from a directory (non-recursive).
125
126 Args:
127 directory (str): Directory path to scan
128
129 Returns:
130 list: List of service file paths
131 """
132 service_files = []
133
134 # Look for all files in the directory (not subdirectories)
135 for item in os.listdir(directory):

137 if os.path.isfile(item_path):

143def expand_directory_dependencies(directory_dep_path, service_dir):
144 """
145 Expand directory-based dependencies by reading files in the specified directory.
146
147 Args:
148 directory_dep_path (str): Path relative to the service file
149 service_dir (str): Directory containing the service file
150
151 Returns:
152 list: List of service names from the directory
153 """
154 # Resolve the actual directory path
155 if directory_dep_path.startswith('/'):
156 actual_dir = directory_dep_path
157 else:
158 actual_dir = os.path.join(service_dir, directory_dep_path)
159
160 services = []
161 if os.path.isdir(actual_dir):
162 for item in os.listdir(actual_dir):
163 if not item.startswith('.'): # Skip hidden files/directories
164 # Extract service name without parameters (before @ symbol)
165 base_service_name = item.split('@')[0] if '@' in item else item
166 services.append(base_service_name)
167
168 return services

211def add_dependency_edges(service_name, dependencies, service_dir, nodes, edges):
212 """Add dependency edges to the graph."""
213 # Add all dependency types that create edges (hard dependencies)
214 for dep_type in ['depends-on', 'depends-ms', 'waits-for']:
215 for dep in dependencies[dep_type]:
216 if dep.endswith('.d'):
217 # This is a directory dependency
218 expanded_deps = expand_directory_dependencies(dep[:-2], service_dir)
219 for expanded_dep in expanded_deps:
220 # Always add the dependency as a node, regardless of whether it exists in scanned files
221 if expanded_dep not in nodes:
222 nodes[expanded_dep] = ('unknown', False) # Default type for missing dependencies
223 edges.add((service_name, expanded_dep, dep_type))
224 else:
225 # Regular dependency - always add as a node even if not found in scanned files
226 if dep not in nodes:
227 nodes[dep] = ('unknown', False) # Default type for missing dependencies
228 edges.add((service_name, dep, dep_type))
229
230 # Handle 'after' dependencies (ordering, but still represent as edges)
231 for dep in dependencies['after']:
232 if dep not in nodes:
233 nodes[dep] = ('unknown', False) # Default type for missing dependencies
234 edges.add((service_name, dep, 'after'))
235
236 # Handle 'before' dependencies (reverse of after)
237 for dep in dependencies['before']:
238 if dep not in nodes:
239 nodes[dep] = ('unknown', False) # Default type for missing dependencies
240 edges.add((dep, service_name, 'after')) # before is reverse of after

243def build_dependency_graph(service_files):
244 """
245 Build a dependency graph from service files.
246
247 Args:
248 service_files (list): List of service file paths
249
250 Returns:
251 tuple: (nodes, edges) where nodes is a dict mapping service names to types and parametric info and
252 edges is a list of (source, target, dependency_type) tuples
253 """
254 # Dictionary to store service name -> (type, has_parameters)
255 nodes = {}
256 edges = set() # Using set to avoid duplicate edges
257
258 # First pass: collect all service names and their types and parametric information
259 for service_file in service_files:
260 service_info = parse_service_file(service_file)
261 service_name = service_info['name']
262 service_type = service_info['type']
263 has_parameters = service_info['has_parameters']
264 nodes[service_name] = (service_type, has_parameters)
265
266 # Second pass: parse dependencies and build edges
267 for service_file in service_files:
268 service_info = parse_service_file(service_file)
269 service_name = service_info['name']
270 service_dir = os.path.dirname(service_file)
271 dependencies = service_info['dependencies']
272
273 add_dependency_edges(service_name, dependencies, service_dir, nodes, edges)
274
275 return nodes, list(edges)

278def generate_hiearch_format(nodes, edges, target_services=None):
279 """
280 Generate hiearch YAML format from nodes and edges.
281
282 Args:
283 nodes (dict): Dictionary of service names to (their types, has_parameters)
284 edges (list): List of (source, target, dependency_type) tuples
285 target_services (list): List of target services to create a dedicated view for (None if all services)
286
287 Returns:
288 str: hiearch YAML format string
289 """
290 hiearch_data = {
291 'nodes': [],
292 'edges': [],
293 'views': []
294 }
295
296 # Add nodes with styles based on service type
297 for service_name, (service_type, has_parameters) in sorted(nodes.items()):
298 # Add "@" suffix to service name if it supports parameters
299 display_name = service_name + '@' if has_parameters else service_name
300 style = get_style_for_service_type(service_type)
301 hiearch_data['nodes'].append({
302 'id': [display_name, service_name], # [label, unique id] - label shows @ if parametric, id stays the same
303 'style_notag': style
304 })
305
306 # Add edges with different styles based on dependency type
307 for source, target, dep_type in edges:
308 edge_style = get_edge_style_for_dependency_type(dep_type)
309 # Use actual node IDs in the link, not display names (the display names are for the nodes themselves)
310 hiearch_data['edges'].append({
311 'link': [source, target], # Use actual node IDs, not display names
312 'style': edge_style
313 })
314
315 # Create view(s) based on whether specific services were selected
316 if target_services:
317 # Create a dedicated view for the selected services and their dependencies
318 hiearch_data['views'].append({
319 'id': 'dinit_service_selection',
320 'nodes': target_services,
321 'neighbours': 'recursive_out',
322 'style': 'hh_dinit_service_view'
323 })
324
325 # Also create an "all services" view
326 hiearch_data['views'].append({
327 'id': 'dinit_service_all',
328 'style': 'hh_dinit_service_view',
329 'tags': ['default']
330 })
331 else:
332 # Create a default view for all services
333 hiearch_data['views'].append({
334 'id': 'dinit_service_all',
335 'style': 'hh_dinit_service_view',
336 'tags': ['default']
337 })
338
339 # Use safe_dump to avoid issues with special YAML characters
340 return yaml.safe_dump(hiearch_data, default_flow_style=False, allow_unicode=True)
