50 class MessageWriter
51 {
52 public:
53 using DataContainer = boost::container::flat_map<std::string, double>;
54
55 public:
56 DataContainer data_;
57 uint64_t current_timestamp_;
58 uint64_t prev_timestamp_;
59 pjmsg_mcap_wrapper::Message message_;
60 pjmsg_mcap_wrapper::Writer mcap_writer_;
61
62 public:
63 MessageWriter()
64 {
65 current_timestamp_ = 0;
66 prev_timestamp_ = 0;
67 mcap_writer_.initialize("graphite_to_mcap.mcap", "/graphite_to_mcap");
68 message_.reserve(500);
69 }
70
71 void add(const std::string &name, const double value, const uint64_t timestamp)
72 {
73 if (current_timestamp_ != timestamp)
74 {
75 if (not data_.empty())
76 {
77 if (data_.size() != message_.size())
78 {
79 message_.bumpVersion();
80 message_.resize(data_.size());
81 }
82
83 std::size_t index = 0;
84 bool version_updated = false;;
85 for (const DataContainer::value_type &entry : data_)
86 {
87 if (entry.first != message_.names()[index])
88 {
89 message_.bumpVersion();
90 version_updated = true;
91 }
92 if (version_updated)
93 {
94 message_.names()[index] = entry.first;
95 }
96 message_.values()[index] = entry.second;
97
98 ++index;
99 }
100
101 mcap_writer_.write(message_);
102 mcap_writer_.flush();
103
104 data_.clear();
105 }
106
107 prev_timestamp_ = current_timestamp_;
108 current_timestamp_ = timestamp;
109 message_.setStamp(timestamp);
110 }
111
112 data_[name] = value;
113 }
114 };
115} // namespace

179 }
180 else

183 }
184 for (const std::regex &match_pattern : parameters.microsecond_to_percent_regex_)
185 {
186 if (std::regex_match(name, match_pattern))
187 {
188 percent_flag = true;
189 break;
190 }
191 }
192 token_start = token_end + 1;
193 }
194
195 token_end = line.find(' ', token_start);
196 if ((std::string::npos == token_end) or (token_start >= token_end))
197 {
198 throw std::runtime_error("Cannot read metric value.");
199 }
200 const double value = boost::lexical_cast<double>(line.substr(token_start, token_end - token_start));
201 token_start = token_end + 1;
202
203
204 if (token_start >= line.size())
205 {
206 throw std::runtime_error("Cannot read metric stamp.");
207 }
208 const uint64_t timestamp =
209 boost::lexical_cast<uint64_t>(line.substr(token_start, line.size() - token_start - 1))
210 * std::nano::den;
211
212
213 message_writer.add(name, value, timestamp);
214 if (percent_flag) // microseconds to percent
215 {
216 double percent_value = 0.0;
217 if (message_writer.prev_timestamp_ > 0 && message_writer.prev_timestamp_ < timestamp)
218 {
219 // (metric_value / (duration_in_nanoseconds / 1000)) * 100
220 const uint64_t factor = (timestamp - message_writer.prev_timestamp_) / 100;
221 percent_value = value / static_cast<double>(factor);
222 }
223 message_writer.add(name + "_percent", percent_value, timestamp);
224 }
225 }
226 }
227 }
228 catch (const std::exception &e)
229 {

231 return (EXIT_FAILURE);
232 }

234 return (EXIT_SUCCESS);
