64 INTROMETRY_HIDDEN std::string normalizeId(const std::string &input_id)
65 {
66 std::string result;
67 result.resize(input_id.size());
68 std::transform(
69 input_id.cbegin(),
70 input_id.cend(),
71 result.begin(),
72 [](unsigned char c) -> unsigned char
73 {
74 c = std::tolower(c);
75 if (intrometry_private::backend::valid_chars.find(static_cast<char>(c)) == std::string::npos)
76 {
77 return ('_');
78 }
79 return (c);
80 });
81
82 const std::size_t first_non_underscore = result.find_first_not_of('_');
83 if (first_non_underscore == std::string::npos)
84 {
85 result = "";
86 }
87 else
88 {
89 if (first_non_underscore > 0)
90 {
91 result = result.substr(first_non_underscore);
92 }
93 }
94
95 return (result);
96 }

142 {
143 // the clock is monotonic, so this is always >= 0
144 const std::chrono::nanoseconds time_diff = std::chrono::steady_clock::now() - pimpl_->time_threshold_;
145 pimpl_->time_threshold_ += (time_diff / pimpl_->step_ + 1) * pimpl_->step_;
