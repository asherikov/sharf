109 // additions and removals of sources are exclusive

116 {
117 // source.arilesDefaultID()
118 // this is redundant and less reliable than type_index
119 // since it is provided by the user
120
121 // id
122 // this is also provided by the user and is also unreliable,
123 // but in general provides extra information than type_index
124 // and should be added to hash

128 std::string getUniqueId(const std::string &id)
129 {
130 const typename CollisionMap::iterator collision_counter_it = collision_counters_.find(id);
131 if (collision_counters_.end() == collision_counter_it)
132 {
133 collision_counters_[id] = 0;
134 return (id);
135 }
136
137 ++collision_counter_it->second;
138 return (str_concat(id, "_intrometry", std::to_string(collision_counter_it->second)));
139 }
140
141 public:
142 void tryVisit(const std::function<void(t_Value &)> visitor)
143 {
144 if (sources_mutex_.try_lock_shared())
145 {
146 for (std::pair<const Key, SourceWithMutex> &source : sources_)
147 {
148 source.second.tryVisit(visitor);
149 }
150
151 sources_mutex_.unlock();
152 }
153 }

157 {

174 bool tryVisit(
175 const std::string &id,
176 const ariles2::DefaultBase &source,
177 const std::function<void(t_Value &)> visitor)
178 {
179 if (sources_mutex_.try_lock_shared())
180 {
181 const typename SourceMap::iterator source_it = sources_.find(getKey(id, source));
182
183 if (sources_.end() == source_it)
184 {
185 sources_mutex_.unlock();
186 return (false);
187 }
188
189 source_it->second.tryVisit(visitor);
190 sources_mutex_.unlock();
191 }
192 return (true);
193 }
