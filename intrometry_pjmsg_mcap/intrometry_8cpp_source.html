33 class NameValueContainer : public ariles2::namevalue2::NameValueContainer
34 {
35 public:
36 plotjuggler_msgs::msg::StatisticsNames names_;
37 plotjuggler_msgs::msg::StatisticsValues values_;
38 std::size_t previous_size_ = 0;
39 bool new_names_version_ = false;
40
41
42 public: // ariles stuff
43 void finalize(const bool persistent_structure, const uint64_t timestamp, uint32_t &names_version)
44 {
45 const int32_t sec = static_cast<int32_t>(timestamp / std::nano::den);
46 const uint32_t nanosec = timestamp % std::nano::den;
47
48 names_.header().stamp().sec(sec);
49 values_.header().stamp().sec(sec);
50 names_.header().stamp().nanosec(nanosec);
51 values_.header().stamp().nanosec(nanosec);
52
53 // we cannot know for sure that the names have not changed
54 // without comparing all the names, do our best
55 if (not persistent_structure or previous_size_ != size())
56 {
57 names_.names_version(names_version);
58 values_.names_version(names_version);
59 new_names_version_ = true;
60 ++names_version;
61 }
62
63 previous_size_ = size();
64 }
65
66 std::string &name(const std::size_t index)
67 {
68 return (names_.names()[index]);
69 }
70 double &value(const std::size_t index)
71 {
72 return (values_.values()[index]);
73 }
74
75 void reserve(const std::size_t size)
76 {
77 names_.names().reserve(size);
78 values_.values().reserve(size);
79 }
80 [[nodiscard]] std::size_t size() const
81 {
82 return (names_.names().size());
83 }
84 void resize(const std::size_t size)
85 {
86 names_.names().resize(size);
87 values_.values().resize(size);
88 }
89 };

95 class McapCDRWriter
96 {
97 protected:
98 template <class t_Message>
99 class Channel
100 {
101 protected:
102 mcap::Message message_;
103 eprosima::fastcdr::CdrSizeCalculator cdr_size_calculator_;
104
105 protected:
106 uint32_t getSize(const t_Message &message)
107 {
108 size_t current_alignment{ 0 };
109 return (cdr_size_calculator_.calculate_serialized_size(message, current_alignment)
110 + 4u /*encapsulation*/);
111 }
112
113 public:
114 Channel() : cdr_size_calculator_(eprosima::fastcdr::CdrVersion::XCDRv1)
115 {
116 }
117
118 void initialize(mcap::McapWriter &writer, const std::string_view &msg_topic)
119 {
120 mcap::Schema schema(
121 intrometry_private::pjmsg_mcap::Message<t_Message>::type,
122 "ros2msg",
123 intrometry_private::pjmsg_mcap::Message<t_Message>::schema);
124 writer.addSchema(schema);
125
126 mcap::Channel channel(msg_topic, "ros2msg", schema.id);
127 writer.addChannel(channel);
128
129 message_.channelId = channel.id;
130 }
131
132 void write(mcap::McapWriter &writer, std::vector<std::byte> &buffer, const t_Message &message)
133 {
134 buffer.resize(getSize(message));
135 message_.data = buffer.data();
136
137 {
138 eprosima::fastcdr::FastBuffer cdr_buffer(
139 reinterpret_cast<char *>(buffer.data()), buffer.size()); // NOLINT
140 eprosima::fastcdr::Cdr ser(
141 cdr_buffer, eprosima::fastcdr::Cdr::DEFAULT_ENDIAN, eprosima::fastcdr::CdrVersion::XCDRv1);
142 ser.set_encoding_flag(eprosima::fastcdr::EncodingAlgorithmFlag::PLAIN_CDR);
143
144 ser.serialize_encapsulation();
145 ser << message;
146 ser.set_dds_cdr_options({ 0, 0 });
147
148 message_.dataSize = ser.get_serialized_data_length();
149 }
150
151 message_.logTime = intrometry::backend::now();
152 message_.publishTime = message_.logTime;
153
154
155 const mcap::Status res = writer.write(message_);
156 if (not res.ok())
157 {
158 throw std::runtime_error(
159 intrometry::backend::str_concat("Failed to write a message: ", res.message));
160 }
161 }
162 };
163
164 public:
165 std::tuple<Channel<plotjuggler_msgs::msg::StatisticsNames>, Channel<plotjuggler_msgs::msg::StatisticsValues>>
166 channels_;
167
168 std::vector<std::byte> buffer_;
169 mcap::McapWriter writer_;
170
171 public:
172 ~McapCDRWriter()
173 {
174 writer_.close();
175 }
176
177 void initialize(const std::filesystem::path &filename, const std::string &topic_prefix)
178 {
179 {
180 const mcap::McapWriterOptions options = mcap::McapWriterOptions("ros2msg");
181 const mcap::Status res = writer_.open(filename.native(), options);
182 if (not res.ok())
183 {
184 throw std::runtime_error(intrometry::backend::str_concat(
185 "Failed to open ", filename.native(), " for writing: ", res.message));
186 }
187 }
188
189 std::get<Channel<plotjuggler_msgs::msg::StatisticsNames>>(channels_).initialize(
190 writer_, intrometry::backend::str_concat(topic_prefix, "/names"));
191
192 std::get<Channel<plotjuggler_msgs::msg::StatisticsValues>>(channels_).initialize(
193 writer_, intrometry::backend::str_concat(topic_prefix, "/values"));
194 }
195
196 template <class t_Message>
197 void write(const t_Message &message)
198 {
199 std::get<Channel<t_Message>>(channels_).write(writer_, buffer_, message);
200 }
201 };
202} // namespace

206 class WriterWrapper
207 {
208 public:
209 const std::string id_; // NOLINT
210 ariles2::namevalue2::Writer::Parameters writer_parameters_;
211 std::shared_ptr<NameValueContainer> data_;
212 ariles2::namevalue2::Writer writer_;
213
214 bool serialized_;
215
216 public:
217 WriterWrapper(
218 const ariles2::DefaultBase &source,
219 std::string id,
220 const bool persistent_structure,
221 uint32_t &names_version)
222 : id_(std::move(id)), data_(std::make_shared<NameValueContainer>()), writer_(data_)
223 {
224 writer_parameters_ = writer_.getDefaultParameters();
225 if (persistent_structure)
226 {
227 writer_parameters_.persistent_structure_ = true;
228 }
229
230 // write to allocate memory
231 ariles2::apply(writer_, source, id_);
232 data_->finalize(writer_parameters_.persistent_structure_, 0, names_version);
233
234 serialized_ = true; // do not serialize on assignment
235 }
236
237 void serialize(McapCDRWriter &mcap_writer)
238 {
239 if (not serialized_)
240 {
241 if (data_->new_names_version_)
242 {
243 mcap_writer.write(data_->names_);
244 data_->new_names_version_ = false;
245 }
246
247 mcap_writer.write(data_->values_);
248 serialized_ = true;
249 }
250 }
251
252 void write(const ariles2::DefaultBase &source, const uint64_t timestamp, uint32_t &names_version)
253 {
254 ariles2::apply(writer_, source, id_);
255 data_->finalize(writer_parameters_.persistent_structure_, timestamp, names_version);
256 serialized_ = false;
257 }
258 };
259} // namespace

298 class Implementation
299 {
300 protected:
301 McapCDRWriter mcap_writer_;
302
303 public:
304 uint32_t names_version_;
305
306 intrometry::backend::SourceContainer<WriterWrapper> sources_;
307 tut::thread::Supervisor<> thread_supervisor_;
308
309 public:
310 Implementation(const std::filesystem::path &directory, const std::string &sink_id, const std::size_t rate)
311 {
312 names_version_ = intrometry::backend::getRandomUInt32();
313
314 const std::string node_id = intrometry::backend::normalizeId(sink_id);
315 const std::string random_id = intrometry::backend::getRandomId(8);
316 const std::string topic_prefix =
317 intrometry::backend::str_concat("/intrometry/", node_id.empty() ? random_id : node_id);
318
319 if (not directory.empty())
320 {
321 std::filesystem::create_directories(directory);
322 }
323 const std::filesystem::path filename = directory
324 / intrometry::backend::str_concat(
325 node_id,
326 node_id.empty() ? "" : "_",
327 random_id,
328 "_",
329 intrometry::backend::getDateString(),
330 ".mcap");
331
332 mcap_writer_.initialize(filename, topic_prefix);
333
334 thread_supervisor_.add(
335 tut::thread::Parameters(
336 tut::thread::Parameters::Restart(/*attempts=*/100, /*sleep_ms=*/50),
337 tut::thread::Parameters::TerminationPolicy::IGNORE,
338 tut::thread::Parameters::ExceptionPolicy::CATCH),
339 &Implementation::spin,
340 this,
341 rate);
342 }
343
344 virtual ~Implementation()
345 {
346 thread_supervisor_.stop();
347 }
348
349
350 void spin(const std::size_t rate)
351 {
352 intrometry::backend::RateTimer timer(rate);
353
354 if (timer.valid())
355 {
356 timer.start();
357
358 while (not thread_supervisor_.isInterrupted())
359 {
360 sources_.tryVisit([this](WriterWrapper &writer) { writer.serialize(mcap_writer_); });
361
362 timer.step();
363 }
364 }
365 else
366 {
367 thread_supervisor_.log("Incorrect spin rate");
368 }
369 thread_supervisor_.interrupt();
370 }
371 };
372} // namespace intrometry::pjmsg_mcap::sink
