76 class WriterWrapper
77 {
78 public:
79 const std::string id_; // NOLINT
80 ariles2::namevalue2::Writer::Parameters writer_parameters_;
81 std::shared_ptr<NameValueContainer> data_;
82 ariles2::namevalue2::Writer writer_;
83
84 bool serialized_;
85
86 public:
87 WriterWrapper(
88 const ariles2::DefaultBase &source,
89 std::string id,
90 const bool persistent_structure,
91 uint32_t &names_version)
92 : id_(std::move(id)), data_(std::make_shared<NameValueContainer>()), writer_(data_)
93 {
94 writer_parameters_ = writer_.getDefaultParameters();
95 if (persistent_structure)
96 {
97 writer_parameters_.persistent_structure_ = true;
98 }
99
100 // write to allocate memory
101 ariles2::apply(writer_, source, id_);
102 data_->finalize(writer_parameters_.persistent_structure_, 0, names_version);
103
104 serialized_ = true; // do not serialize on assignment
105 }
106
107 void serialize(pjmsg_mcap_wrapper::Writer &mcap_writer)
108 {
109 if (not serialized_)
110 {
111 mcap_writer.write(data_->message_);
112 serialized_ = true;
113 }
114 }
115
116 void write(const ariles2::DefaultBase &source, const uint64_t timestamp, uint32_t &names_version)
117 {
118 ariles2::apply(writer_, source, id_);
119 data_->finalize(writer_parameters_.persistent_structure_, timestamp, names_version);
120 serialized_ = false;
121 }
122 };
123} // namespace

162 class Implementation
163 {
164 protected:
165 pjmsg_mcap_wrapper::Writer mcap_writer_;
166
167 public:
168 uint32_t names_version_;
169
170 intrometry::backend::SourceContainer<WriterWrapper> sources_;
171 tut::thread::Supervisor<> thread_supervisor_;
172
173 public:
174 Implementation(const std::filesystem::path &directory, const std::string &sink_id, const std::size_t rate)
175 {
176 names_version_ = intrometry::backend::getRandomUInt32();
177
178 const std::string node_id = intrometry::backend::normalizeId(sink_id);
179 const std::string random_id = intrometry::backend::getRandomId(8);
180 const std::string topic_prefix =
181 intrometry::backend::str_concat("/intrometry/", node_id.empty() ? random_id : node_id);
182
183 if (not directory.empty())
184 {
185 std::filesystem::create_directories(directory);
186 }
187 const std::filesystem::path filename = directory
188 / intrometry::backend::str_concat(
189 node_id,
190 node_id.empty() ? "" : "_",
191 random_id,
192 "_",
193 intrometry::backend::getDateString(),
194 ".mcap");
195
196 mcap_writer_.initialize(filename, topic_prefix);
197
198 thread_supervisor_.add(
199 tut::thread::Parameters(
200 tut::thread::Parameters::Restart(/*attempts=*/100, /*sleep_ms=*/50),
201 tut::thread::Parameters::TerminationPolicy::IGNORE,
202 tut::thread::Parameters::ExceptionPolicy::CATCH),
203 &Implementation::spin,
204 this,
205 rate);
206 }
207
208 virtual ~Implementation()
209 {
210 thread_supervisor_.stop();
211 }
212
213
214 void spin(const std::size_t rate)
215 {
216 intrometry::backend::RateTimer timer(rate);
217
218 if (timer.valid())
219 {
220 timer.start();
221
222 while (not thread_supervisor_.isInterrupted())
223 {
224 sources_.tryVisit([this](WriterWrapper &writer) { writer.serialize(mcap_writer_); });
225
226 timer.step();
227 }
228 }
229 else
230 {
231 thread_supervisor_.log("Incorrect spin rate");
232 }
233 thread_supervisor_.interrupt();
234 }
235 };
236} // namespace intrometry::pjmsg_mcap::sink
