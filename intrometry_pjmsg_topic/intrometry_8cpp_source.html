36 class NameValueContainer : public ariles2::namevalue2::NameValueContainer
37 {
38 public:
39 NamesMsg names_;
40 ValuesMsg values_;
41 std::size_t previous_size_ = 0;
42 bool new_names_version_ = false;
43
44 public: // ariles stuff
45 void finalize(const bool persistent_structure, const rclcpp::Time &timestamp, uint32_t &names_version)
46 {
47 names_.header.stamp = timestamp;
48 values_.header.stamp = timestamp;
49
50 // we cannot know for sure that the names have not changed
51 // without comparing all the names, do our best
52 if (not persistent_structure or previous_size_ != size())
53 {
54 names_.names_version = names_version;
55 values_.names_version = names_version;
56 new_names_version_ = true;
57 ++names_version;
58 }
59
60 previous_size_ = size();
61 }
62
63 std::string &name(const std::size_t index)
64 {
65 return (names_.names[index]);
66 }
67 double &value(const std::size_t index)
68 {
69 return (values_.values[index]);
70 }
71
72 void reserve(const std::size_t size)
73 {
74 names_.names.reserve(size);
75 values_.values.reserve(size);
76 }
77 [[nodiscard]] std::size_t size() const
78 {
79 return (names_.names.size());
80 }
81 void resize(const std::size_t size)
82 {
83 names_.names.resize(size);
84 values_.values.resize(size);
85 }
86 };

92 class WriterWrapper
93 {
94 public:
95 const std::string id_; // NOLINT
96 ariles2::namevalue2::Writer::Parameters writer_parameters_;
97 std::shared_ptr<NameValueContainer> data_;
98 ariles2::namevalue2::Writer writer_;
99
100 bool published_;
101
102
103 public:
104 WriterWrapper(
105 const ariles2::DefaultBase &source,
106 std::string id,
107 const bool persistent_structure,
108 uint32_t &names_version)
109 : id_(std::move(id)), data_(std::make_shared<NameValueContainer>()), writer_(data_)
110 {
111 writer_parameters_ = writer_.getDefaultParameters();
112 if (persistent_structure)
113 {
114 writer_parameters_.persistent_structure_ = true;
115 }
116
117 // write to allocate memory
118 ariles2::apply(writer_, source, id_);
119 data_->finalize(writer_parameters_.persistent_structure_, rclcpp::Time(0), names_version);
120
121 published_ = true; // do not publish on assignment
122 }
123
124
125 void publish(const NamesPublisherPtr &names_sink, const ValuesPublisherPtr &values_sink)
126 {
127 if (not published_)
128 {
129 if (data_->new_names_version_)
130 {
131 names_sink->publish(data_->names_);
132 data_->new_names_version_ = false;
133 }
134 values_sink->publish(data_->values_);
135 published_ = true;
136 }
137 }
138
139
140 void write(const ariles2::DefaultBase &source, const rclcpp::Time &timestamp, uint32_t &names_version)
141 {
142 ariles2::apply(writer_, source, id_);
143 data_->finalize(writer_parameters_.persistent_structure_, timestamp, names_version);
144 published_ = false;
145 }
146 };
147} // namespace

212 class Implementation
213 {
214 public:
215 std::shared_ptr<rclcpp::Node> node_;
216
217 protected:
218 NamesPublisherPtr names_publisher_;
219 ValuesPublisherPtr values_publisher_;
220
221 public:
222 uint32_t names_version_;
223
224 intrometry::backend::SourceContainer<WriterWrapper> sources_;
225 tut::thread::Supervisor<ROSLogger> thread_supervisor_;
226
227 public:
228 Implementation(const std::string &sink_id, const std::size_t rate)
229 {
230 names_version_ = intrometry::backend::getRandomUInt32();
231
232 const std::string node_id = intrometry::backend::normalizeId(sink_id);
233 const std::string random_id = intrometry::backend::getRandomId(8);
234 const std::string topic_prefix =
235 intrometry::backend::str_concat("intrometry/", node_id.empty() ? random_id : node_id);
236
237 node_ = std::make_shared<rclcpp::Node>(
238 intrometry::backend::str_concat("intrometry_", node_id, "_", random_id),
239 // try to be stealthy and use minimal resources
240 rclcpp::NodeOptions()
241 .enable_topic_statistics(false)
242 .start_parameter_services(false)
243 .start_parameter_event_publisher(false)
244 .append_parameter_override("use_sim_time", false)
245 .use_clock_thread(false)
246 .enable_rosout(false));
247 thread_supervisor_.initializeLogger(node_);
248
249 names_publisher_ = node_->create_publisher<NamesMsg>(
250 intrometry::backend::str_concat(topic_prefix, "/names"),
251 rclcpp::QoS(/*history_depth=*/20).reliable().transient_local());
252 values_publisher_ = node_->create_publisher<ValuesMsg>(
253 intrometry::backend::str_concat(topic_prefix, "/values"),
254 rclcpp::QoS(/*history_depth=*/20).best_effort().durability_volatile());
255
256
257 thread_supervisor_.add(
258 tut::thread::Parameters(
259 tut::thread::Parameters::Restart(/*attempts=*/100, /*sleep_ms=*/50),
260 tut::thread::Parameters::TerminationPolicy::IGNORE,
261 tut::thread::Parameters::ExceptionPolicy::CATCH),
262 &Implementation::spin,
263 this,
264 rate);
265 }
266
267 virtual ~Implementation()
268 {
269 thread_supervisor_.stop();
270 }
271
272
273 void spin(const std::size_t rate)
274 {
275 intrometry::backend::RateTimer timer(rate);
276
277 if (timer.valid())
278 {
279 timer.start();
280
281 while (rclcpp::ok() and not thread_supervisor_.isInterrupted())
282 {
283 sources_.tryVisit([this](WriterWrapper &writer)
284 { writer.publish(names_publisher_, values_publisher_); });
285 rclcpp::spin_some(node_);
286
287 timer.step();
288 }
289 }
290 else
291 {
292 thread_supervisor_.log("Incorrect spin rate");
293 }
294 thread_supervisor_.interrupt();
295 }
296 };
297 } // namespace sink
298} // namespace intrometry::pjmsg_topic
