28 template <class t_Message>
29 class Subscription
30 {
31 public:
32 typename rclcpp::Subscription<t_Message>::SharedPtr subscription_;
33 std::atomic<std::size_t> counter_ = 0;
34
35 public:
36 void subscribe(std::shared_ptr<rclcpp::Node> &node, const std::string &topic)
37 {
38 subscription_ = node->create_subscription<t_Message>(
39 topic,
40 rclcpp::QoS(/*history_depth=*/10).best_effort(),
41 [this](const t_Message &msg)
42 {
43 std::stringstream output;
44 plotjuggler_msgs::msg::to_block_style_yaml(msg, output);
45 std::cerr << "---------------------------\n";
46 std::cerr << output.str();
47 std::cerr << std::flush;
48 ++counter_;
49 });
50 }
51
52 [[nodiscard]] bool atLeastOne() const
53 {
54 return (counter_ > 0);
55 }
56 };

104 void spin()
105 {
106 const std::chrono::nanoseconds step(std::nano::den / 1000);
107
108 std::chrono::time_point<std::chrono::steady_clock> time_threshold = std::chrono::steady_clock::now();
109 while (rclcpp::ok() and not isThreadSupervisorInterrupted())
110 {
111 rclcpp::spin_some(node_);
112 time_threshold += step;
113 std::this_thread::sleep_until(time_threshold);
114 }
