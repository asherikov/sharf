110 class Scheduling
111 {
112 public:
113 int priority_; /// thread priority
114 int policy_; /// thread policy
115 bool ignore_failures_; /// proceed if scheduling parameters could not be set
116
117 public:
118 explicit Scheduling(
119 const int priority = 0, // NOLINT
120 const int policy = SCHED_FIFO, // NOLINT
121 const bool ignore_failures = true)
122 {
123 priority_ = priority;
124 policy_ = policy;
125 ignore_failures_ = ignore_failures;
126 }

128
129 bool apply(const std::thread::native_handle_type &&handle) const // NOLINT return value can be ignored
130 {
131 if (SCHED_FIFO != policy_ or 0 != priority_) // custom parameters
132 {
133 sched_param sched_params;
134 sched_params.sched_priority = priority_;
135
136 if (0 != pthread_setschedparam(handle, policy_, &sched_params))
137 {
138 return (false);
139 }
140 }
141 return (true);
142 }

213 template <class t_Logger, class t_Function, class... t_Args>
214 void startOnce(Supervisor<t_Logger> *supervisor, t_Function &&function, t_Args &&...args)
215 {
216 switch (parameters_.exception_policy_)
217 {
218 case Parameters::ExceptionPolicy::PASS:
219 std::bind(function, std::forward<t_Args>(args)...)();
220 break;
221
222 case Parameters::ExceptionPolicy::CATCH:
223 try
224 {
225 std::bind(function, std::forward<t_Args>(args)...)();
226 }
227 catch (const std::exception &e)
228 {
229 supervisor->log("Supervisor / intercepted thread exception: ", e.what());
230 }
231 break;
232
233 default:
234 supervisor->log("Supervisor error: unknown exception handling type");
235 break;
236 }
237 }

239
240 template <class t_Logger, class t_Function, class... t_Args>
241 void startLoop(Supervisor<t_Logger> *supervisor, t_Function &&function, t_Args &&...args)
242 {
243 bool started = false;
244 for (std::size_t attempt = 0; parameters_.restart_.isOk(attempt) and not supervisor->isInterrupted();
245 ++attempt)
246 {
247 if (started)
248 {
249 parameters_.restart_.wait();
250 if (supervisor->isInterrupted())
251 {
252 break;
253 }
254
255 supervisor->log(
256 "Supervisor / restarting thread: ",
257 attempt + 1,
258 " / ",
259 parameters_.restart_.isUnlimited() ? 0 : parameters_.restart_.attempts_);
260 }
261 else
262 {
263 started = true;
264 }
265
266 startOnce(supervisor, function, std::forward<t_Args>(args)...);
267 }
268 }

271 template <class t_Logger, class t_Function, class... t_Args>
272 void startThread(Supervisor<t_Logger> *supervisor, t_Function &&function, t_Args &&...args)
273 {
274 if (parameters_.restart_.isEnabled())
275 {
276 startLoop(supervisor, function, std::forward<t_Args>(args)...);
277 }
278 else
279 {
280 startOnce(supervisor, function, std::forward<t_Args>(args)...);
281 }
282
283
284 switch (parameters_.termination_policy_)
285 {
286 case Parameters::TerminationPolicy::KILLALL:
287 supervisor->interrupt();
288 supervisor->drop(self_);
289 break;
290
291 case Parameters::TerminationPolicy::IGNORE:
292 supervisor->drop(self_);
293 break;
294
295 default:
296 supervisor->log("Supervisor error: unknown termination handling type");
297 break;
298 }
299 }

301
302 public:
303 template <class t_Logger, class... t_Args>
304 void start(
305 Supervisor<t_Logger> *supervisor,
306 Reference self,
307 const Parameters &&parameters,
308 t_Args &&...args)
309 {
310 self_ = self;
311 parameters_ = parameters;
312 thread_ = std::thread(
313 &Thread::startThread<t_Logger, t_Args...>, this, supervisor, std::forward<t_Args>(args)...);
314
315 if (not parameters_.scheduling_.apply(thread_.native_handle()))
316 {
317 supervisor->log("Supervisor error: could not configure custom thread scheduling.");
318 if (not parameters_.scheduling_.ignore_failures_)
319 {
320 std::terminate();
321 }
322 }
323 }

327 if (thread_.joinable())
328 {
329 thread_.join();
330 }

336 /// Thread supervisor
337 template <class t_Logger>
338 class Supervisor : public t_Logger
339 {
340 friend class Thread;
341 THREAD_SUPERVISOR_DISABLE_CLASS_COPY(Supervisor)
342
343 protected:
344 enum class Status
345 {
346 UNDEFINED = 0,
347 ACTIVE = 1,
348 INTERRUPTED = 2
349 };
350
351
352 protected:
353 std::atomic<Status> status_;
354
355 std::mutex threads_mutex_;
356 Thread::List threads_;
357
358 std::mutex terminated_threads_mutex_;
359 std::list<Thread::Reference> terminated_threads_;
360
361
362 protected:
363 bool wait(const std::size_t wait_ms)
364 {
365 const std::size_t sleep_ms = 10;
366
367 {
368 const std::lock_guard<std::mutex> lock(threads_mutex_);
369 std::size_t counter = 0;
370 for (;;)
371 {
372 if (counter * sleep_ms < wait_ms)
373 {
374 {
375 const std::lock_guard<std::mutex> terminated_lock(terminated_threads_mutex_);
376
377 for (const Thread::Reference &thread_ref : terminated_threads_)
378 {
379 thread_ref->join();
380 threads_.erase(thread_ref);
381 }
382 terminated_threads_.clear();
383 }
384
385 if (not threads_.empty())
386 {
387 std::this_thread::sleep_for(std::chrono::milliseconds(sleep_ms));
388 ++counter;
389 }
390 else
391 {
392 return (true);
393 }
394 }
395 else
396 {
397 // cppcheck-suppress ignoredReturnValue
398 log("Threads did not terminate in the given time after request.");
399 return (false);
400 }
401 }
402 }
403 }
404
405
406 bool empty()
407 {
408 const std::lock_guard<std::mutex> lock(threads_mutex_);
409 return (threads_.empty());
410 }
411
412
413 void drop(const std::list<Thread>::iterator &item)
414 {
415 const std::lock_guard<std::mutex> lock(terminated_threads_mutex_);
416 terminated_threads_.push_back(item);
417 }
418
419
420 public:
421 using t_Logger::log;
422
423
424 Supervisor()
425 {
426 status_ = Status::UNDEFINED;
427 }
428
429
430 ~Supervisor()
431 {
432 if (Status::ACTIVE == status_)
433 {
434 // - throwing in destructor -> terminate(), unless noexcept(false) is set.
435 // - noexcept(false) however interferes with other stuff in derived classes.
436 // - this check is intended to detect API misuse and should not fail under
437 // normal conditions.
438 // cppcheck-suppress ignoredReturnValue
439 log("Destructor of Supervisor is reached with active status,"
440 "which means that interrupt() method has not been called.");
441 std::terminate();
442 }
443
444 if (not empty())
445 {
446 // cppcheck-suppress ignoredReturnValue
447 log("Destructor of Supervisor is reached with some threads still running.");
448 std::terminate();
449 }
450 }
451
452
453 /// Interrupt execution, see @ref isInterrupted
454 void interrupt()
455 {
456 status_ = Status::INTERRUPTED;
457 }
458
459
460 /// Check if interrupted, child threads should stop when true (pass supervisor as a parameter).
461 [[nodiscard]] bool isInterrupted() const
462 {
463 return (Status::INTERRUPTED == status_);
464 }
465
466
467 /// Interrupt and wait for children to stop
468 bool stop(const std::size_t wait_ms = 10000)
469 {
470 interrupt();
471 return (wait(wait_ms));
472 }
473
474
475 /// Add a thread: (<thread parameters>, <function pointer>, <function parameters>)
476 template <class... t_Args>
477 void add(t_Args &&...args)
478 {
479 if (isInterrupted())
480 {
481 // cppcheck-suppress ignoredReturnValue
482 log("Addition of a thread attempted after interrupt.");
483 std::terminate();
484 }
485 else
486 {
487 status_ = Status::ACTIVE;
488 const std::lock_guard<std::mutex> lock(threads_mutex_);
489 threads_.emplace_back();
490 threads_.back().start(this, --threads_.end(), std::forward<t_Args>(args)...);
491 }
492 }
